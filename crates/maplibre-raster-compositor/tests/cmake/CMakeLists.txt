function(assert_composed_hooks)
  get_target_property(sources maplibre_native_c_objects SOURCES)
  foreach(file IN ITEMS custom_layer.cpp vulkan.cpp vulkan_retirement.cpp)
    set(found FALSE)
    foreach(source IN LISTS sources)
      if(source MATCHES "/${file}$")
        set(found TRUE)
      endif()
    endforeach()
    if(NOT found)
      message(FATAL_ERROR "Composed hooks did not inject ${file}")
    endif()
  endforeach()
  list(LENGTH sources count)
  if(NOT count EQUAL 5)
    message(FATAL_ERROR "Unexpected hook source count: ${count}")
  endif()
endfunction()
cmake_language(DEFER CALL assert_composed_hooks)
